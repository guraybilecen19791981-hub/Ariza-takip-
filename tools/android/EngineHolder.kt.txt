package __PKG__

import android.content.Context
import android.content.Intent
import android.net.Uri
import android.os.Build
import android.provider.Settings
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.embedding.engine.FlutterEngineCache
import io.flutter.embedding.engine.dart.DartExecutor
import io.flutter.plugin.common.MethodChannel

object EngineHolder {
    private const val ENGINE_ID = "ariza_main"
    private const val PREFS = "ariza_prefs"
    private const val KEY_ENABLED = "keepalive_enabled"

    @Synchronized
    fun get(context: Context): FlutterEngine {
        val cached = FlutterEngineCache.getInstance().get(ENGINE_ID)
        if (cached != null) return cached

        val app = context.applicationContext
        val engine = FlutterEngine(app)
        MethodChannel(engine.dartExecutor.binaryMessenger, "ariza/keepalive").setMethodCallHandler { call, result ->
            when (call.method) {
                "start" -> {
                    setEnabled(app, true)
                    KeepAliveService.start(app)
                    result.success(true)
                }
                "stop" -> {
                    setEnabled(app, false)
                    app.stopService(Intent(app, KeepAliveService::class.java))
                    result.success(true)
                }
                "batterySettings" -> {
                    openBatterySettings(app)
                    result.success(true)
                }
                else -> result.notImplemented()
            }
        }
        engine.dartExecutor.executeDartEntrypoint(DartExecutor.DartEntrypoint.createDefault())
        FlutterEngineCache.getInstance().put(ENGINE_ID, engine)
        return engine
    }

    fun isEnabled(context: Context): Boolean {
        return context.getSharedPreferences(PREFS, Context.MODE_PRIVATE).getBoolean(KEY_ENABLED, false)
    }

    private fun setEnabled(context: Context, value: Boolean) {
        context.getSharedPreferences(PREFS, Context.MODE_PRIVATE).edit().putBoolean(KEY_ENABLED, value).apply()
    }

    private fun openBatterySettings(context: Context) {
        try {
            val intent = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                Intent(Settings.ACTION_REQUEST_IGNORE_BATTERY_OPTIMIZATIONS, Uri.parse("package:" + context.packageName))
            } else {
                Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS, Uri.parse("package:" + context.packageName))
            }
            intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            context.startActivity(intent)
        } catch (e: Exception) {
            val intent = Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS, Uri.parse("package:" + context.packageName))
            intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            context.startActivity(intent)
        }
    }
}
